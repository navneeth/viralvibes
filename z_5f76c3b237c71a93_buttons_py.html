58def paste_button(target_id: str) -> Button: 

59 """Paste button with icon swap and premium styling.""" 

60 status_id = f"{target_id}_status" 

61 onclick = f""" 

62 const btn = this, input = document.getElementById('{target_id}'), status = document.getElementById('{status_id}'); 

63 btn.disabled = true; 

64 navigator.clipboard.readText() 

65 .then(text => {{ 

66 input.value = text.trim(); 

67 input.dispatchEvent(new Event('input', {{ bubbles: true }})); 

68 status.textContent = '✓ Pasted'; 

69 status.className = 'text-green-600 text-xs font-semibold'; 

70 setTimeout(() => status.textContent = '', 1500); 

71 }}) 

72 .catch(() => {{ 

73 status.textContent = '✗ Paste failed'; 

74 status.className = 'text-red-600 text-xs font-semibold'; 

75 setTimeout(() => status.textContent = '', 2000); 

76 }}) 

77 .finally(() => btn.disabled = false); 

78 """ 

79 

80 return Button( 

81 UkIcon("clipboard", cls="w-4 h-4"), # Slightly smaller icon 

82 type="button", 

83 onclick=onclick, 

84 cls=( 

85 # ✅ SIZING 

86 "w-9 h-9 " # Explicit 36×36px (slightly larger for comfort) 

87 "flex items-center justify-center " 

88 # ✅ STYLING 

89 "text-gray-400 hover:text-red-600 " 

90 "focus:outline-none focus:ring-2 focus:ring-red-500/30 " 

91 "disabled:opacity-50 disabled:cursor-not-allowed " 

92 # ✅ INTERACTIONS 

93 "transition-all duration-200 " 

94 "hover:bg-red-50 rounded-md " 

95 "active:scale-95 " 

96 # ✅ RESPONSIVE 

97 "flex-shrink-0" # Never shrinks below 36×36px 

98 ), 

99 title="Paste from clipboard", 

100 aria_label="Paste from clipboard",

208def YtIcon(variant: str = "branded", size: int = 20) -> NotStr: 

209 """Inline SVG YouTube play-button icon. 

210 

211 variant="branded" — red rectangle, white triangle (for light/transparent bg). 

212 variant="mono" — white evenodd punch-out shape (for use inside a red button). 

213 Both meet the 20 px minimum size requirement from the YouTube API branding guidelines. 

214 """ 

215 if variant == "mono": 

216 return NotStr( 

217 f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24"' 

218 f' width="{size}" height="{size}" aria-hidden="true" focusable="false">' 

219 f'<path fill="#fff" fill-rule="evenodd" d="{_YT_PUNCHOUT}"/>' 

220 f"</svg>" 

221 ) 

222 # branded (default) 

223 return NotStr( 

224 f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24"' 

225 f' width="{size}" height="{size}" aria-hidden="true" focusable="false">' 

226 f'<path fill="#FF0000" d="{_YT_RECT}"/>' 

227 f'<path fill="#fff" d="{_YT_TRIANGLE}"/>' 

228 f"</svg>" 

229 )

232def YoutubeChannelButton( 

233 channel_url: str, 

234 *, 

235 variant: str = "solid", 

236 as_button: bool = False, 

237 onclick_js: str = "", 

238 size: str = "sm", 

239 extra_cls: str = "", 

240) -> ...: 

241 """Branded YouTube channel button — compliant with YouTube API branding guidelines. 

242 

243 variant="solid" — red (#FF0000) pill with white icon + text. Used on 

244 profile pages and compare cards where the button is a 

245 primary action standing alone. 

246 variant="ghost" — transparent background with branded (red+white) icon 

247 and red label. Used inside card footers that are 

248 themselves wrapped in an <a>, so nested <a> is invalid. 

249 Pass as_button=True to render a <button> with onclick. 

250 

251 size="sm" → py-1.5 px-3 text-xs (card footer, compact contexts) 

252 size="md" → py-2 px-4 text-sm (profile page, compare card) 

253 

254 as_button=True renders a <button> element (for contexts inside an existing 

255 <a>) instead of an <a>. Pair with onclick_js for the click handler. 

256 """ 

257 pad = "px-3 py-1.5 text-xs" if size == "sm" else "px-4 py-2 text-sm" 

258 icon_size = 20 # always ≥20px per YouTube branding guidelines 

259 

260 if variant == "ghost": 

261 icon = YtIcon("branded", size=icon_size) 

262 cls = ( 

263 f"inline-flex items-center gap-1.5 {pad} rounded-lg font-semibold " 

264 f"text-red-600 hover:text-red-700 dark:text-red-400 dark:hover:text-red-300 " 

265 f"bg-transparent border-0 cursor-pointer transition-colors {extra_cls}" 

266 ) 

267 content = (icon, Span("YouTube")) 

268 if as_button: 

269 return Button( 

270 *content, 

271 type="button", 

272 onclick=onclick_js, 

273 cls=cls, 

274 aria_label="Open YouTube channel", 

275 ) 

276 return A( 

277 *content, 

278 href=channel_url, 

279 target="_blank", 

280 rel="noopener noreferrer", 

281 cls=cls, 

282 aria_label="Open YouTube channel", 

283 ) 

284 

285 # solid (default) 

286 icon = YtIcon("mono", size=icon_size) 

287 cls = ( 

288 f"inline-flex items-center gap-1.5 {pad} rounded-lg font-semibold " 

289 f"text-white no-underline transition-colors {extra_cls}" 

290 ) 

291 # YouTube brand red — do not substitute with Tailwind red-600 (#DC2626) 

292 style = "background:#FF0000;" 

293 hover_script = ( 

294 "this.style.background='#CC0000'" 

295 ",this.onmouseleave=function(){this.style.background='#FF0000'}" 

296 ) 

297 if as_button: 

298 return Button( 

299 icon, 

300 Span("YouTube"), 

301 type="button", 

302 onclick=onclick_js, 

303 cls=cls, 

304 style=style, 

305 onmouseenter=hover_script, 

306 aria_label="Open YouTube channel", 

307 ) 

308 return A( 

309 icon, 

310 Span("YouTube"), 

311 href=channel_url, 

312 target="_blank", 

313 rel="noopener noreferrer", 

314 cls=cls, 

315 style=style, 

316 onmouseenter=hover_script, 

317 aria_label="Open YouTube channel", 

318 )

321def SignUpNudge( 

322 feature: str = "this feature", 

323 benefit: str = "sign in to unlock full access", 

324 return_url: str = "/", 

325) -> Alert: 

326 """Inline soft sign-up prompt — shown in place of a hard auth wall. 

327 

328 Stays on the current page; never redirects. Converts better than a 

329 wall because users have already seen value before being asked to sign in. 

330 

331 Built on MonsterUI's Alert + DivFullySpaced primitives so it inherits 

332 theme tokens and DaisyUI's alert accessibility role automatically. 

333 

334 Args: 

335 feature: Short label for what is locked, e.g. "full playlist analysis". 

336 benefit: Value proposition line shown under the heading. 

337 return_url: Where to redirect after successful sign-in. 

338 """ 

339 return Alert( 

340 DivFullySpaced( 

341 DivHStacked( 

342 Div( 

343 UkIcon("lock", cls="size-5"), 

344 cls="size-10 rounded-full flex items-center justify-center bg-background/60 flex-shrink-0", 

345 ), 

346 Div( 

347 P(f"Sign in to unlock {feature}", cls="font-semibold text-sm"), 

348 P(benefit, cls="text-xs opacity-70 mt-0.5"), 

349 ), 

350 ), 

351 A( 

352 UkIcon("log-in", cls="size-4"), 

353 Span("Sign in with Google"), 

354 href=f"/login?return_url={return_url}", 

355 cls=(ButtonT.primary, "inline-flex items-center gap-2 flex-shrink-0 text-sm"), 

356 ), 

357 ), 

358 cls=(AlertT.info, "my-6"), 

359 )
