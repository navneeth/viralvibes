1""" 

2Blog data layer — Post model, post loading, and markdown rendering. 

3 

4Pattern adapted from https://github.com/jackhogan/personal-site: 

5 - YAML frontmatter parsed by python-frontmatter 

6 - Markdown rendered via monsterui's render_md / FrankenRenderer 

7 - Internal site links (known prefixes + fragment anchors) open same-tab; 

8 all other links open in a new tab with rel="noopener noreferrer" 

9"""

28class Post: 

29 """A single markdown blog post with YAML frontmatter. 

30 

31 Frontmatter keys: 

32 title (str) — display title 

33 date (date) — publication date used for sort order 

34 excerpt (str) — one-paragraph teaser shown on the index 

35 tags (list) — optional tag list e.g. ["strategy", "data"] 

36 placeholder (bool) — when true the post renders the coming-soon SVG 

37 instead of the markdown body 

38 """ 

39 

40 def __init__(self, path: Path) -> None: 

41 import frontmatter # lazy — not installed in the worker / test env 

42 

43 self.path = path 

44 self.slug = path.stem 

45 

46 post = frontmatter.load(str(path)) 

47 self.content: str = post.content 

48 self.meta: dict = post.metadata 

49 

50 self.title: str = self.meta.get("title", self.slug.replace("-", " ").title()) 

51 

52 raw_date = self.meta.get("date") 

53 self.date: Optional[date] = raw_date if isinstance(raw_date, date) else None 

54 self.datestr: str = self.date.strftime("%d %b %Y") if self.date else "" 

55 

56 self.excerpt: str = self.meta.get("excerpt", "") 

57 raw_tags = self.meta.get("tags", []) 

58 if isinstance(raw_tags, list): 

59 self.tags: list[str] = [str(t) for t in raw_tags] 

60 else: 

61 if raw_tags: 

62 logger.warning( 

63 "blog: %s — 'tags' must be a YAML list, got %r; ignoring", path, raw_tags 

64 ) 

65 self.tags = [] 

66 raw_placeholder = self.meta.get("placeholder", False) 

67 if not isinstance(raw_placeholder, bool): 

68 logger.warning( 

69 "blog: %s — 'placeholder' must be a boolean (true/false), got %r; defaulting to false", 

70 path, 

71 raw_placeholder, 

72 ) 

73 raw_placeholder = False 

74 self.placeholder: bool = raw_placeholder

82def get_posts(published_only: bool = False) -> list[Post]: 

83 """Return posts from ``posts/``, sorted newest-first. 

84 

85 Args: 

86 published_only: When True, exclude posts with ``placeholder: true``. 

87 """ 

88 if not _POSTS_DIR.exists(): 

89 return [] 

90 posts: list[Post] = [] 

91 for path in _POSTS_DIR.glob("*.md"): 

92 try: 

93 posts.append(Post(path)) 

94 except Exception: 

95 logger.exception("blog: failed to load post %s — skipping", path) 

96 posts.sort(key=lambda p: p.date or date.min, reverse=True) 

97 

98 return [p for p in posts if not p.placeholder] if published_only else posts

101def get_post(slug: str) -> Optional[Post]: 

102 """Return the post matching *slug*, or None.""" 

103 path = _POSTS_DIR / f"{slug}.md" 

104 if not path.exists(): 

105 return None 

106 try: 

107 return Post(path) 

108 except Exception: 

109 logger.exception("blog: failed to load post %s", slug) 

110 return None

138def _make_renderer(): 

139 """Build a FrankenRenderer subclass that adds HTMX attrs to internal links.""" 

140 try: 

141 from monsterui.all import FrankenRenderer 

142 import mistletoe 

143 

144 # Known in-app route prefixes — only these get same-tab link behaviour. 

145 _INTERNAL_PREFIXES = ("/blog", "/creators", "/creator", "/dashboard", "/compare", "/lists") 

146 

147 class _SiteRenderer(FrankenRenderer): 

148 """Extends FrankenRenderer: internal site links open in the same tab; 

149 external links get target='_blank' and rel='noopener noreferrer'.""" 

150 

151 def render_link(self, token): 

152 href = token.target or "" 

153 inner = self.render_inner(token) 

154 title_attr = f' title="{token.title}"' if getattr(token, "title", None) else "" 

155 is_internal = href.startswith("#") or any( 

156 href.startswith(p) for p in _INTERNAL_PREFIXES 

157 ) 

158 if is_internal: 

159 return ( 

160 f'<a href="{href}"{title_attr} ' 

161 f'class="text-primary underline underline-offset-2 hover:opacity-80">' 

162 f"{inner}</a>" 

163 ) 

164 return ( 

165 f'<a href="{href}"{title_attr} ' 

166 f'target="_blank" rel="noopener noreferrer" ' 

167 f'class="text-primary underline underline-offset-2 hover:opacity-80">' 

168 f"{inner}</a>" 

169 ) 

170 

171 return _SiteRenderer 

172 except Exception: 

173 logger.warning("blog: FrankenRenderer not available, falling back to render_md default") 

174 return None

180def from_md(content: str) -> "object": 

181 """Render *content* as styled markdown, returning a FastHTML Div.""" 

182 from fasthtml.common import Div, NotStr 

183 

184 try: 

185 from monsterui.all import render_md 

186 

187 global _renderer_cls 

188 if _renderer_cls is None: 

189 _renderer_cls = _make_renderer() 

190 

191 kwargs: dict = {"class_map_mods": _MD_CLASS_MODS} 

192 if _renderer_cls is not None: 

193 from functools import partial 

194 

195 kwargs["renderer"] = partial(_renderer_cls) 

196 

197 rendered = render_md(content, **kwargs) 

198 return Div(NotStr(_PROSE_STYLE), rendered, cls="prose-content max-w-none") 

199 except Exception: 

200 logger.exception("blog: markdown render failed") 

201 # Graceful fallback — show raw text rather than a 500 

202 from fasthtml.common import P 

203 

204 return Div(P(content, cls="text-sm font-mono text-muted-foreground whitespace-pre-wrap"))
