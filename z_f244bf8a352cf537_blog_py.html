23def compute_reading_time_minutes(content: str) -> int: 

24 """Return reading time in whole minutes, rounded up, minimum 1. 

25 

26 Uses 200 wpm — a conservative average that errs on the side of 

27 slightly over-estimating, consistent with common blog UX practice. 

28 ``ceil`` prevents underestimation: 250 words → 2 min, not 1. 

29 """ 

30 return max(1, math.ceil(len(content.split()) / 200))

185def _tag_nav(all_tags: list[str], active_tag: str | None) -> Div: 

186 """Horizontal pill row for filtering by tag. 'All' clears the filter.""" 

187 _base = "text-xs px-3 py-1 rounded-full border font-medium transition-colors no-underline " 

188 _active = _base + "bg-primary text-primary-foreground border-primary" 

189 _inactive = ( 

190 _base + "border-border text-muted-foreground hover:border-primary hover:text-foreground" 

191 ) 

192 

193 pills = [ 

194 A( 

195 "All", 

196 href="/blog", 

197 cls=_active if active_tag is None else _inactive, 

198 ) 

199 ] + [ 

200 A( 

201 tag, 

202 href=f"/blog?tag={quote_plus(tag)}", 

203 cls=_active if tag == active_tag else _inactive, 

204 ) 

205 for tag in all_tags 

206 ] 

207 return Div(*pills, cls="flex flex-wrap gap-2 mb-8")

210def _post_row(post) -> A: 

211 """One post entry for the index listing.""" 

212 is_placeholder = post.placeholder 

213 

214 title_el = Span( 

215 post.title, 

216 cls=( 

217 "text-base font-semibold text-foreground group-hover:text-primary transition-colors" 

218 if not is_placeholder 

219 else "text-base font-semibold text-muted-foreground" 

220 ), 

221 ) 

222 

223 meta_parts = [] 

224 if post.datestr: 

225 meta_parts.append(Span(post.datestr, cls="text-xs text-muted-foreground")) 

226 if is_placeholder: 

227 meta_parts.append( 

228 Span( 

229 "Coming Soon",

232 ) 

233 else: 

234 _rm = compute_reading_time_minutes(post.content) 

235 meta_parts.append(Span(f"{_rm} min read", cls="text-xs text-muted-foreground")) 

236 meta_parts.extend([_tag_pill(t) for t in post.tags]) 

237 

238 meta_row = Div(*meta_parts, cls="flex flex-wrap items-center gap-2 mt-1") 

239 

240 excerpt_el = ( 

241 P(post.excerpt, cls="text-sm text-muted-foreground leading-relaxed mt-1 line-clamp-2") 

242 if post.excerpt and not is_placeholder 

243 else None 

244 )

246 return A( 

247 Div( 

248 title_el, 

249 meta_row, 

250 excerpt_el, 

251 cls="py-4 border-b border-border last:border-0",

253 href=f"/blog/{post.slug}", 

254 cls="group block no-underline hover:no-underline",

258def blog_index_content(posts: list, active_tag: str | None = None) -> Div: 

259 """Blog index page body. 

260 

261 When *active_tag* is set, only posts carrying that tag are listed and the 

262 tag nav highlights the active filter. Placeholder posts are always shown 

263 regardless of tag (they display a Coming Soon badge, not actual tags). 

264 """ 

265 if not posts: 

266 return blog_coming_soon_content() 

267 

268 # Collect unique tags from published posts (preserve insertion order across posts) 

269 seen: dict[str, None] = {} 

270 for p in posts: 

271 if not p.placeholder: 

272 for t in p.tags: 

273 seen[t] = None 

274 all_tags = list(seen) 

275 

276 # Filter: placeholders always visible; published posts filtered by tag 

277 if active_tag and active_tag in seen: 

278 visible = [p for p in posts if p.placeholder or active_tag in p.tags] 

279 else: 

280 active_tag = None # ignore unknown tags 

281 visible = posts 

282 

283 published = sum(1 for p in posts if not p.placeholder) 

284 pending = len(posts) - published 

285 

286 header = Div( 

287 P( 

288 "ViralVibes Blog", 

289 cls="text-xs font-mono uppercase tracking-[0.18em] text-blue-600 mb-3", 

290 ), 

291 H1( 

292 "Creator Marketing Intelligence", 

293 cls=( 

294 "text-4xl font-bold tracking-tight mb-4 " 

295 "bg-gradient-to-br from-foreground via-foreground to-foreground/60 " 

296 "bg-clip-text text-transparent" 

297 ), 

298 ), 

299 P( 

300 "Data-driven takes on YouTube creator strategy, campaign optimisation, " 

301 "and the metrics that actually matter.", 

302 cls="text-base text-muted-foreground leading-relaxed max-w-2xl", 

303 ), 

304 cls="mb-10", 

305 ) 

306 

307 post_list = Div(*[_post_row(p) for p in visible]) 

308 

309 tag_nav_el = _tag_nav(all_tags, active_tag) if all_tags else None 

310 

311 coming_soon_note = ( 

312 Div( 

313 P( 

314 f"{pending} more article{'s' if pending != 1 else ''} in progress — check back soon.", 

315 cls="text-sm text-muted-foreground text-center", 

316 ), 

317 cls="mt-8 py-4 border-t border-border", 

318 ) 

319 if pending > 0 

320 else None 

321 ) 

322 

323 return Div( 

324 header, 

325 tag_nav_el, 

326 post_list, 

327 coming_soon_note, 

328 cls="max-w-2xl mx-auto px-4 py-12", 

329 )

396_RSS_DATE_FMT = "%a, %d %b %Y 00:00:00 +0000"

399def build_rss_feed(posts: list, site_url: str) -> str: 

400 """Return a complete RSS 2.0 XML string for all published (non-placeholder) posts. 

401 

402 Args: 

403 posts: Full post list (placeholders are automatically excluded). 

404 site_url: Absolute base URL, e.g. ``"https://www.viralvibes.fyi"``. 

405 """ 

406 

407 def _item(post) -> str: 

408 pub_date = ( 

409 f"\n <pubDate>{post.date.strftime(_RSS_DATE_FMT)}</pubDate>" if post.date else "" 

410 ) 

411 description = escape(post.excerpt) if post.excerpt else "" 

412 link = escape(f"{site_url}/blog/{post.slug}") 

413 tags = "".join(f"\n <category>{escape(t)}</category>" for t in post.tags) 

414 return ( 

415 f" <item>\n" 

416 f" <title>{escape(post.title)}</title>\n" 

417 f" <link>{link}</link>\n" 

418 f' <guid isPermaLink="true">{link}</guid>\n' 

419 f" <description>{description}</description>" 

420 f"{pub_date}{tags}\n" 

421 f" </item>" 

422 ) 

423 

424 published = [p for p in posts if not p.placeholder] 

425 items_xml = "\n".join(_item(p) for p in published) 

426 

427 return ( 

428 '<?xml version="1.0" encoding="UTF-8"?>\n' 

429 '<rss version="2.0" xmlns:atom="http://www.w3.org/2005/Atom">\n' 

430 " <channel>\n" 

431 " <title>ViralVibes Blog</title>\n" 

432 f" <link>{site_url}/blog</link>\n" 

433 f' <atom:link href="{site_url}/rss.xml" rel="self" type="application/rss+xml"/>\n' 

434 " <description>Data-driven takes on YouTube creator strategy and campaign optimisation.</description>\n" 

435 " <language>en-gb</language>\n" 

436 f"{items_xml}\n" 

437 " </channel>\n" 

438 "</rss>" 

439 )
